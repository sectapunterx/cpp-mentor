# tests/CMakeLists.txt — GoogleTest pulled in reproducibly via FetchContent.
include(FetchContent)
FetchContent_Declare(
  googletest
  GIT_REPOSITORY https://github.com/google/googletest.git
  GIT_TAG v1.15.2
)
FetchContent_MakeAvailable(googletest)

add_executable({{NS}}_{{PROJECT}}_tests
    {{MODULE}}/{{FILE}}_test.cc
)
target_link_libraries({{NS}}_{{PROJECT}}_tests
    PRIVATE {{NS}}::{{PROJECT}} GTest::gtest_main)
target_compile_options({{NS}}_{{PROJECT}}_tests PRIVATE -Wall -Wextra -Wpedantic)

include(GoogleTest)
gtest_discover_tests({{NS}}_{{PROJECT}}_tests)
